      "mov 0x8(%0), %%edx            \n"
      "mov 0x8(%0), %%ecx            \n"
      "cmp %%r8d, %%ecx              \n"
      "cmovge %%r8d, %%edx           \n"
      "cmovl %%r8d, %%ecx            \n"
      "mov %%ecx, %%r8d              \n"
      "mov (%0), %%eax               \n"
      "cmp %%eax, %%r8d              \n"
      "cmovge %%eax, %%r8d           \n"

      "mov %%ecx, 0x8(%0)            \n"

      "mov %%eax, (%0)               \n"

      :

}
